/** * 華南高商校園整潔評分系統 * * 功能： * 1. 照片上傳至 Google Drive * 2. 紀錄寫入 Google 試算表 * 3. JSONP 查詢紀錄 * 4. 管理者密碼驗證 * 5. 管理者刪除紀錄及照片 * * 必須設定的指令碼屬性： * ADMIN_PASSWORD = 自訂管理者密碼 */ const FOLDER_NAME = "整潔資料測試"; const SPREADSHEET_NAME = "整潔評分資料庫"; const HEADERS = [ "時間", "班級", "缺失項目", "相片連結" ]; /* ===================================================== 網頁 GET：提供 JSONP 查詢紀錄 ===================================================== */ function doGet(e) { try { const action = e && e.parameter ? e.parameter.action : ""; if (action === "list") { const records = getRecords_(); const result = { result: "success", records: records }; const callback = e.parameter.callback; // HTML 使用 JSONP，因此回傳 JavaScript 格式 if (callback) { if (!/^[a-zA-Z_$][0-9a-zA-Z_$]*$/.test(callback)) { throw new Error("無效的 JSONP callback"); } return ContentService .createTextOutput( callback + "(" + JSON.stringify(result) + ");" ) .setMimeType(ContentService.MimeType.JAVASCRIPT); } return jsonResponse_(result); } return jsonResponse_({ result: "success", message: "校園整潔評分系統 GAS 正常運作" }); } catch (err) { const result = { result: "error", error: err.message }; const callback = e && e.parameter ? e.parameter.callback : ""; if (callback && /^[a-zA-Z_$][0-9a-zA-Z_$]*$/.test(callback)) { return ContentService .createTextOutput( callback + "(" + JSON.stringify(result) + ");" ) .setMimeType(ContentService.MimeType.JAVASCRIPT); } return jsonResponse_(result); } } /* ===================================================== 網頁 POST：照片上傳、登入驗證、刪除紀錄 ===================================================== */ function doPost(e) { try { const data = getPostData_(e); const action = String(data.action || ""); // 1. 管理者登入驗證 if (action === "login") { const auth = verifyAdminPassword_( data.password || data.adminPassword || "" ); return jsonResponse_({ result: auth.ok ? "success" : "error", authenticated: auth.ok, message: auth.message }); } // 2. 管理者刪除紀錄 if (action === "delete") { deleteRecord_(data); return jsonResponse_({ result: "success", message: "紀錄刪除成功" }); } // 3. 一般照片上傳 if (data.imageData) { uploadRecord_(data); return jsonResponse_({ result: "success", message: "照片及紀錄上傳成功" }); } throw new Error("無法辨識的操作或缺少照片資料"); } catch (err) { return jsonResponse_({ result: "error", error: err.message }); } } /* ===================================================== 解析 POST 資料 ===================================================== */ function getPostData_(e) { if (!e) { throw new Error("沒有收到 POST 資料"); } // 支援 application/json 或 text/plain 傳送的 JSON if (e.postData && e.postData.contents) { const contents = e.postData.contents; try { return JSON.parse(contents); } catch (err) { // 若不是 JSON，改讀取表單參數 } } // 支援 application/x-www-form-urlencoded if (e.parameter) { return e.parameter; } throw new Error("POST 資料格式錯誤"); } /* ===================================================== 照片上傳及寫入試算表 ===================================================== */ function uploadRecord_(data) { const className = String(data.className || "").trim(); const issue = String(data.issue || "").trim(); const timestamp = String(data.timestamp || "").trim(); const fileName = String(data.fileName || "").trim(); const imageData = String(data.imageData || ""); if (!className) { throw new Error("缺少班級資料"); } if (!imageData) { throw new Error("缺少照片資料"); } // 取得 Base64 const base64Match = imageData.match( /^data:image\/([a-zA-Z0-9.+-]+);base64,(.+)$/ ); if (!base64Match) { throw new Error("照片格式錯誤，必須為 Base64 圖片"); } const imageType = base64Match[1].toLowerCase(); if (!["jpeg", "jpg", "png", "webp"].includes(imageType)) { throw new Error("不支援的照片格式：" + imageType); } const bytes = Utilities.base64Decode(base64Match[2]); if (!bytes || bytes.length === 0) { throw new Error("照片內容為空"); } const mimeType = imageType === "jpg" ? "image/jpeg" : "image/" + imageType; const blob = Utilities.newBlob( bytes, mimeType, fileName || ("整潔評分_" + Date.now() + ".jpg") ); // 取得或建立 Drive 資料夾 const folder = getOrCreateFolder_(); // 儲存照片 const file = folder.createFile(blob); // 取得試算表 const sheet = getDataSheet_(); // 寫入資料 sheet.appendRow([ timestamp || Utilities.formatDate( new Date(), "Asia/Taipei", "yyyy/MM/dd HH:mm:ss" ), className, issue, file.getUrl() ]); return { fileId: file.getId(), fileUrl: file.getUrl() }; } /* ===================================================== 查詢全部紀錄 ===================================================== */ function getRecords_() { const sheet = getDataSheet_(); const lastRow = sheet.getLastRow(); if (lastRow < 2) { return []; } const values = sheet .getRange(2, 1, lastRow - 1, 4) .getDisplayValues(); const records = []; values.forEach(function(row, index) { const timestamp = row[0]; const className = row[1]; const issue = row[2]; const fileUrl = row[3]; // 略過完全空白的資料列 if (!timestamp && !className && !issue && !fileUrl) { return; } const fileId = extractDriveFileId_(fileUrl); records.push({ rowNumber: index + 2, timestamp: timestamp, className: className, issue: issue, fileUrl: fileUrl, fileId: fileId, thumbnailUrl: fileId ? "https://drive.google.com/thumbnail?id=" + encodeURIComponent(fileId) + "&sz=w800" : "" }); }); // 最新紀錄排在前面 records.reverse(); return records; } /* ===================================================== 管理者密碼驗證 ===================================================== */ function verifyAdminPassword_(inputPassword) { const savedPassword = PropertiesService .getScriptProperties() .getProperty("ADMIN_PASSWORD"); if (!savedPassword) { return { ok: false, message: "尚未設定管理者密碼，請聯絡系統管理員" }; } const input = String(inputPassword || ""); if (!constantTimeEquals_(input, savedPassword)) { return { ok: false, message: "管理者密碼錯誤" }; } return { ok: true, message: "管理者驗證成功" }; } /* ===================================================== 固定時間字串比較 ===================================================== */ function constantTimeEquals_(a, b) { a = String(a); b = String(b); let diff = a.length ^ b.length; const maxLength = Math.max(a.length, b.length); for (let i = 0; i < maxLength; i++) { const charA = i < a.length ? a.charCodeAt(i) : 0; const charB = i < b.length ? b.charCodeAt(i) : 0; diff |= charA ^ charB; } return diff === 0; } /* ===================================================== 刪除紀錄及照片 ===================================================== */ function deleteRecord_(data) { // 每次刪除都必須重新驗證密碼 const auth = verifyAdminPassword_( data.password || data.adminPassword || "" ); if (!auth.ok) { throw new Error(auth.message); } const rowNumber = Number(data.rowNumber); if ( !Number.isInteger(rowNumber) || rowNumber < 2 ) { throw new Error("無效的資料列編號"); } const sheet = getDataSheet_(); if (rowNumber > sheet.getLastRow()) { throw new Error("找不到指定的資料列"); } // 重新讀取資料，避免刪錯列 const row = sheet .getRange(rowNumber, 1, 1, 4) .getDisplayValues()[0]; const savedTimestamp = String(row[0] || ""); const savedClassName = String(row[1] || ""); const savedIssue = String(row[2] || ""); const savedFileUrl = String(row[3] || ""); // 比對前端送來的資料，避免資料列已變動卻刪除錯誤紀錄 if ( savedClassName !== String(data.className || "") || savedTimestamp !== String(data.timestamp || "") || savedIssue !== String(data.issue || "") ) { throw new Error("資料內容已變動，為避免誤刪，請重新載入紀錄"); } // 優先使用試算表內的照片連結取得檔案 ID const fileId = extractDriveFileId_(savedFileUrl); // 刪除試算表資料列 sheet.deleteRow(rowNumber); // 將對應照片移至垃圾桶 if (fileId) { try { DriveApp.getFileById(fileId).setTrashed(true); } catch (err) { // 照片不存在或無刪除權限時，不影響資料列刪除 console.error("照片移至垃圾桶失敗：" + err.message); } } return true; } /* ===================================================== 從 Google Drive 網址取得檔案 ID ===================================================== */ function extractDriveFileId_(url) { const value = String(url || ""); if (!value) { return ""; } let match = value.match(/\/d\/([a-zA-Z0-9_-]+)/); if (match) { return match[1]; } match = value.match(/[?&]id=([a-zA-Z0-9_-]+)/); if (match) { return match[1]; } return ""; } /* ===================================================== 取得或建立 Drive 資料夾 ===================================================== */ function getOrCreateFolder_() { const folders = DriveApp.getFoldersByName(FOLDER_NAME); if (folders.hasNext()) { return folders.next(); } return DriveApp.createFolder(FOLDER_NAME); } /* ===================================================== 取得或建立試算表及工作表 ===================================================== */ function getDataSheet_() { const files = DriveApp.getFilesByName(SPREADSHEET_NAME); let spreadsheet; if (files.hasNext()) { const file = files.next(); spreadsheet = SpreadsheetApp.openById(file.getId()); } else { spreadsheet = SpreadsheetApp.create(SPREADSHEET_NAME); } let sheet = spreadsheet.getSheets()[0]; if (!sheet) { sheet = spreadsheet.insertSheet("工作表1"); } // 若工作表完全空白，建立標題列 if (sheet.getLastRow() === 0) { sheet.appendRow(HEADERS); } return sheet; } /* ===================================================== JSON 回應 ===================================================== */ function jsonResponse_(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } /* ===================================================== 手動測試：確認試算表及資料夾可正常取得 ===================================================== */ function testSystem() { const folder = getOrCreateFolder_(); const sheet = getDataSheet_(); Logger.log("Drive 資料夾：" + folder.getName()); Logger.log("試算表：" + sheet.getParent().getName()); Logger.log("工作表：" + sheet.getName()); Logger.log("目前資料筆數：" + Math.max(0, sheet.getLastRow() - 1)); }
